FUNCTION calculate_eigenvalues(Matrix A, int max_iterations = 1000, float tolerance = 1e-10):
    // 1. Ensure the matrix is square
    IF A.rows() != A.cols():
        THROW Error("Eigenvalues can only be calculated for square matrices")

    // 2. Initialize the working matrix
    Matrix T = A.copy()
    int n = T.rows()

    // 3. Iterative QR process
    FOR iteration FROM 1 TO max_iterations:
        // Perform QR decomposition: T = Q * R
        // Matrix.h provides a QR() method returning a tuple
        (Matrix Q, Matrix R) = T.QR() 

        // Update T by multiplying in reverse: T_new = R * Q
        // This preserves eigenvalues while making T more "upper triangular"
        Matrix next_T = R * Q

        // 4. Convergence Check
        // Check if the off-diagonal elements are close enough to zero
        float off_diagonal_sum = 0
        FOR i FROM 1 TO n-1:
            FOR j FROM 0 TO i-1:
                off_diagonal_sum += abs(next_T(i, j))

        T = next_T

        IF off_diagonal_sum < tolerance:
            BREAK

    // 5. Extract eigenvalues from the diagonal
    List eigenvalues = []
    FOR i FROM 0 TO n-1:
        eigenvalues.append(T(i, i))

    RETURN eigenvalues
